package core

import (
	"testing"
	"time"

	"github.com/keyorixhq/keyorix/internal/storage/models"
)

// FuzzSF_PasswordExpired: PasswordExpired vs its mutated clone. The password
// age is built as days*24h + delta seconds, so a byte-sized delta lands the
// fuzzer exactly on the MaxAgeDays boundary (a raw int64 age never would:
// native Go fuzzing has no comparison tracing).
func FuzzSF_PasswordExpired(f *testing.F) {
	f.Add(uint8(90), int8(1), false, false)
	f.Fuzz(func(t *testing.T, maxAgeDays uint8, deltaSec int8, useCreated, noTimestamps bool) {
		c := NewKeyorixCore(new(MockStorage))
		fixed := time.Date(2026, 6, 4, 12, 0, 0, 0, time.UTC)
		c.now = func() time.Time { return fixed }
		c.passwordPolicy = PasswordPolicy{MaxAgeDays: int(maxAgeDays)}
		age := time.Duration(maxAgeDays)*24*time.Hour + time.Duration(deltaSec)*time.Second
		ts := fixed.Add(-age)
		u := &models.User{}
		switch {
		case noTimestamps:
		case useCreated:
			u.CreatedAt = ts
		default:
			u.PasswordChangedAt = &ts
		}
		orig, mut := c.PasswordExpired(u), c.PasswordExpired__mut(u)
		if orig != mut {
			t.Fatalf("ORIG %v != MUT %v (maxAgeDays=%d deltaSec=%d useCreated=%v)", orig, mut, maxAgeDays, deltaSec, useCreated)
		}
	})
}
