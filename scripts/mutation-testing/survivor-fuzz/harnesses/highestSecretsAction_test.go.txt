package core

import (
	"sort"
	"testing"

	"github.com/keyorixhq/keyorix/internal/storage/models"
)

// FuzzSF_highestSecretsAction: highestSecretsAction vs its mutated clone on the
// same permission list. Each input byte is one permission: bit 7 picks the
// resource (secrets / other), the low bits pick an action from the known
// ranked actions plus an unknown one and the empty string.
func FuzzSF_highestSecretsAction(f *testing.F) {
	f.Add([]byte{0, 1, 2, 3})
	f.Fuzz(func(t *testing.T, b []byte) {
		actions := []string{"", "unknown"}
		for k := range secretsActionRank {
			actions = append(actions, k)
		}
		sort.Strings(actions)
		perms := make([]*models.Permission, 0, len(b))
		for _, x := range b {
			res := "secrets"
			if x&0x80 != 0 {
				res = "other"
			}
			perms = append(perms, &models.Permission{Resource: res, Action: actions[int(x&0x7f)%len(actions)]})
		}
		orig := highestSecretsAction(perms)
		mut := highestSecretsAction__mut(perms)
		if orig != mut {
			t.Fatalf("ORIG %q != MUT %q for %d perms", orig, mut, len(perms))
		}
	})
}
